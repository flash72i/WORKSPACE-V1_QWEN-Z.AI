Спасение рабочей области чата

СКРИПТ ДЛЯ QWEN И Z.AI

Инструкция по запуску скрипта в консоли браузера. Скрипт выгружает файлы рабочей области любого чата на chat.qwen.ai и chat.z.ai: боковая панель, дерево с размерами, один ZIP со структурой папок, прогресс и отмена. Код находится прямо на этой странице — ставить ничего не нужно, только скопировать и вставить в консоль.

1. Как применить

  1. Откройте в браузере chat.qwen.ai или chat.z.ai и войдите в аккаунт.
  2. Откройте чат, у которого есть панель «Рабочая область» с файлами. Адрес страницы должен быть вида chat.qwen.ai/c/<длинный-id> (у z.ai — chat.z.ai/c/<id> или публичная ссылка /s/<id>).
  3. Нажмите F12 (или Ctrl+Shift+I) — откроются инструменты разработчика.
  4. Перейдите на вкладку Console (Консоль).
  5. Если консоль предупреждает про вставку текста — наберите allow и нажмите Enter.
  6. Оставьте консоль открытой и вернитесь на эту страницу. В разделе «3. Скрипт» нажмите «СКОПИРОВАТЬ СКРИПТ» — код уже внутри файла, скачивать ничего не нужно.
  7. Кликните внутрь консоли, вставьте код (Ctrl+V) и нажмите Enter.
  8. В правом верхнем углу страницы появится панель WORKSPACE V1 — такая же, как на макете справа. Её можно перетаскивать за зелёную шапку, кнопки ◀ ▶ прижимают к краям, «—» сворачивает.
  9. В дереве отметьте галочками нужные файлы или папки (галочка на папке выделяет всё внутри). Кнопка ☑ рядом с фильтром отмечает всё, повторное нажатие снимает все отметки. Если файлов очень много, список идёт порциями: прокручивайте вниз — следующие файлы догружаются сами (или нажмите «показать ещё»).
  10. Нажмите «СКАЧАТЬ ZIP» — браузер сохранит один архив со структурой папок. Распакуйте его в любую папку.
  11. Нужны файлы из другого чата? Перейдите в него и нажмите в панели «ОБНОВИТЬ» — заново открывать консоль и вставлять код не нужно.
Скрипт универсален: id чата берётся из адреса страницы, поэтому подходит для любого чата с рабочей областью. После перезагрузки страницы панель исчезает — код нужно вставить снова.
Скрипт ничего не отправляет на сторонние серверы: все запросы идут только к тому сайту, на котором вы его запустили, от имени вашей сессии. Архив собирается прямо в браузере.

2. Если что-то пошло не так

  1. Панель не появилась — проверьте, что вы на странице /c/<id>, после вставки кода нажали Enter и в консоли нет красного текста.
  2. «путей: 0» — у этого чата пустая рабочая область или список ещё не загружен: сверните и раскройте штатную панель файлов сайта и нажмите «ОБНОВИТЬ».
  3. Нет размеров у файлов — в «НАСТРОЙКАХ» выберите «показывать размеры всех файлов» или нажмите «УЗНАТЬ РАЗМЕРЫ СЕЙЧАС». На z.ai размер берётся из ответа их API; «—» остаётся у файлов крупнее 16 МБ и у тех, что не влезли в лимит подсчёта (32 МБ за круг) — нажмите кнопку ещё раз или включите автоопределение.
  4. z.ai: список пуст — песочница спит: нажмите «ОБНОВИТЬ» ещё раз и подождите ~10 секунд.
  5. Имена файлов в архиве исковерканы — архив помечен флагом UTF-8; если архиватор очень старый, откройте его через 7-Zip.
  6. Очень много файлов — при объёме свыше ~900 МБ архив делится на части имя_part2.zip и т. д.; распакуйте все части в одну папку.
  7. Нужна помощь — в панели «НАСТРОЙКИ» → «СДЕЛАТЬ ДИАГНОСТИКУ»: отчёт скопируется в буфер обмена вместе с журналом.

      function done(blob) { if (settled) return; settled = true; clearTimeout(to); resolve(blob); }
      async function fallback() {
        if (settled) return;

        if (!list) { resolve(makeZipSync(entries)); return; }
        const again = await refetchEntries(entries);      // буферы уже ушли в воркер
        if (!again.length) { log('Сборка ZIP сорвалась: воркер недоступен, а перезапросить файлы не удалось.'); resolve(null); return; }
        if (again.length < entries.length) log('  в архив не попало файлов: ' + (entries.length - again.length) + ' (не удалось перезапросить)');

    const one = await bridgeCall('wzr-fetch', { url }, 180000);

  function layoutMain() {
    const bm = $('#bodyMain'), tree = $('#tree'), log = $('#log'), mem = $('#mem'), c = $('#c');

      const s = getComputedStyle(e);
      return e.getBoundingClientRect().height + (parseFloat(s.marginTop) || 0) + (parseFloat(s.marginBottom) || 0);
    };
    let hard = 0;                                            // то, что не сжимается (инструменты, статистика, кнопки)

      if (ch === tree || ch === log || ch === mem || ch === c) continue;

      if (s.display === 'none') continue;
      hard += withMargins(ch);
    }

    let logH = Math.max(0, Math.min(LOG_MAX, avail - hard - memH - cH - TREE_MIN));
    const logEff = logH > 0 ? logH : 10;                      // рамка журнала всё равно занимает ~10px
    let aux = memH + cH;                                     // шкала памяти и строка-подсказка

    if (deficit > 0) {                                       // места мало — уступает память и подсказка

    }

    log.style.flex = '0 1 ' + logH + 'px';
  }

  function renderTree() {

    const vis = visible();
    const limit = TREE_LIMIT || TREE_STEP;
    const shown = vis.length > limit ? vis.slice(0, limit) : vis;
    const rest = vis.length - shown.length;
    const root = { dirs: new Map(), files: [] };
    for (const p of shown) {
      const parts = p.split('/'); let node = root;
      for (let i = 0; i < parts.length - 1; i++) {
        if (!node.dirs.has(parts[i])) node.dirs.set(parts[i], { dirs: new Map(), files: [] });
        node = node.dirs.get(parts[i]);
      }
      node.files.push(parts[parts.length - 1]);

    function build(node, prefix) {
      let s = '';
      const dirEntries = [...node.dirs.entries()];
      const dsum = nm => { const d = curDirSizes.get(prefix + nm + '/'); return d ? d.sum : 0; };
      if (UI.sort === 'alpha') dirEntries.sort((a, b) => a[0].localeCompare(b[0]));
      else if (UI.sort === 'size') dirEntries.sort((a, b) => (dsum(b[0]) - dsum(a[0])) || a[0].localeCompare(b[0]));
      for (const [name, sub] of dirEntries) {
        const dp = prefix + name;
        s += '<details data-dir="' + esc(dp) + '"' + (OPEN_DIRS.has(dp) ? ' open' : '') + '>' +

          '<div style="margin-left:16px">' + build(sub, dp + '/') + '</div></details>';
      }
      const files = node.files.slice();
      const fsize = f => { const v = sizeOf(prefix + f); return v == null ? -1 : v; };
      if (UI.sort === 'alpha') files.sort((a, b) => a.localeCompare(b));
      else if (UI.sort === 'size') files.sort((a, b) => (fsize(b) - fsize(a)) || a.localeCompare(b));

  function decodeText(bytes) {
    /* 1) BOM — самый надёжный признак; 2) объявленная в файле кодировка;
       3) строгий UTF-8; 4) windows-1251 как частый вариант для русских файлов. */
    if (BOM_UTF16LE(bytes)) return { text: new TextDecoder('utf-16le').decode(bytes.subarray(2)), enc: 'utf-16le (BOM)' };
    if (BOM_UTF16BE(bytes)) return { text: new TextDecoder('utf-16be').decode(bytes.subarray(2)), enc: 'utf-16be (BOM)' };
    if (BOM_UTF8(bytes)) return { text: new TextDecoder('utf-8').decode(bytes.subarray(3)), enc: 'utf-8 (BOM)' };
    const head = new TextDecoder('windows-1252').decode(bytes.subarray(0, 4096));
    const declared = (head.match(/charset\s*=\s*["']?\s*([A-Za-z0-9_-]+)/i) || [])[1];
    if (declared && !/^utf-?8$/i.test(declared)) {

      catch (e) { /* заявленная кодировка не подошла — идём дальше */ }
    }
    try { return { text: new TextDecoder('utf-8', { fatal: true }).decode(bytes), enc: 'utf-8' }; }
    catch (e) { return { text: new TextDecoder('windows-1251').decode(bytes), enc: 'windows-1251' }; }

  function htmlAsUtf8(text) {

  let PATH_INDEX = new Set(), ENC_INDEX = new Map(), PATH_IDX_GEN = 0, PATH_IDX_USE = -1;
  function pathIndexes() {
    if (PATH_IDX_USE === PATH_IDX_GEN) return;

  function matchPathInUrl(u) {
    pathIndexes();
    let url; try { url = new URL(u, location.href); } catch (e) { return null; }
    const parts = (url.pathname + url.search).split('/').filter(Boolean);
    for (let i = 0; i < parts.length; i++) {
      const raw = parts.slice(i).join('/');
      if (raw.length <= 4) continue;
      let dec = raw; try { dec = decodeURI(raw); } catch (e) { continue; }
      if (!PATH_INDEX.has(dec)) continue;
      const at = u.indexOf(raw);
      if (at < 0) continue;
      const tok = raw === dec ? '{R}' : (raw === encodeURIComponent(dec) ? '{E}' : '{P}');
      return { path: dec, at, len: raw.length, tok };
    }

      const eq = pair.indexOf('=');
      if (eq < 0) continue;

      const p = ENC_INDEX.get(v);
      if (p && p.length > 4) { const at = u.indexOf(v); if (at >= 0) return { path: p, at, len: v.length, tok: '{E}' }; }
    }
    return null;
  }
